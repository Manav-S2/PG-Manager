function Navbar() {
  return (
    <View style={styles.header}>
      <View style={{ flexDirection: 'row', alignItems: 'center' }}>
        <Text style={styles.logoText}>AVRO</Text>
        <View style={styles.logoAccent} />

          <View style={styles.subtitleDivider} />

        </View>

        <TouchableOpacity style={styles.hostBtn}>
           <Text style={styles.hostBtnText}>Host Controls</Text>
        </TouchableOpacity>

      </View>
    </View>
  );
}

function SidebarItem({ label, icon, badge, isActive, onPress }: any) {
  return (
    <Pressable onPress={onPress} style={[styles.sidebarItem, isActive && styles.sidebarItemActive]}>
      <Ionicons name={icon} size={18} color={isActive ? "#1A1A1A" : "#6E6E73"} />
      <Text style={[styles.sidebarItemText, isActive && styles.sidebarItemTextActive]}>{label}</Text>
      {!!badge && (
        <View style={styles.badge}>
          <Text style={styles.badgeText}>{badge}</Text>
        </View>
      )}
    </Pressable>
  );
}

export default function HostControlPage() {
  const { width } = useWindowDimensions();
  const isWide = width >= 900;
  const [activeItem, setActiveItem] = useState('owner-listing');

  const allItems = SIDEBAR_MENU.flatMap(g => g.items);
  const activeTitle = allItems.find(i => i.id === activeItem)?.label || 'Host Control';

  return (
    <SafeAreaView style={styles.safeArea}>
      <StatusBar barStyle="dark-content" />
      <Navbar />
      
      <View style={styles.container}>
        {isWide && (
          <View style={styles.sidebar}>
            <ScrollView showsVerticalScrollIndicator={false} contentContainerStyle={styles.sidebarContent}>
               <View style={styles.propertySelector}>
                  <Text style={styles.propertySelectorTitle}>Sai Residency</Text>
                  <Text style={styles.propertySelectorMeta}>Kamla Nagar · Delhi</Text>
               </View>

              {SIDEBAR_MENU.map(section => (
                <View key={section.header} style={styles.sidebarSection}>
                  <Text style={styles.sectionHeader}>{section.header}</Text>
                  {section.items.map(item => (
                    <SidebarItem 
                      key={item.id} 
                      {...item} 
                      isActive={activeItem === item.id} 
                      onPress={() => setActiveItem(item.id)} 
                    />
                  ))}
                </View>
              ))}
            </ScrollView>
          </View>
        )}

        <View style={styles.contentArea}>
          {/* Section just below navbar */}
          <View style={styles.pageHeader}>
            <View>
              <Text style={styles.pageTitle}>{activeTitle}</Text>
            </View>
            <View style={styles.pageHeaderRight}>
              <View style={styles.liveDate}>
                <View style={styles.liveDot} />
                <Text style={styles.liveDateText}>Live · Jul 2026</Text>
              </View>
            </View>
          </View>

          {/* Empty Content Area */}
          <ScrollView contentContainerStyle={styles.pageScroll}>
            {/* The rest of the content has been removed as requested. */}
          </ScrollView>
        </View>
      </View>
    </SafeAreaView>
  );
}
